1. Data Table: Search Bar + Serial Numbers

Common Pattern

Standard pattern where user wants quick item lookup and numbered rows without modifying raw database records. 

import runOptions from "json-renderers-run-options";
import definitions from "declarative-definitions";
import { spec } from "spec-repo";

const baseTable = structuredClone(spec.tableSimple);

const enhancedTable = runOptions({
    base: baseTable,
    options: {
        showSearch: true,
        showSerial: true,
        showOptions: false
    },
    optionDefs: definitions.table.simple.options,
    showLog: false
});

console.log("Table with Search and Serial:", enhancedTable);

3. Defining Custom V2 JSONPath Options

Advanced Customization

Creating customized single instructions with custom JSONPath targets evaluated by jsonpathly: 

import runOptions from "json-renderers-run-options";

const customOptionDefs = {
    search: [
        {
            destinationPath: {
                jsonPath: "$..*[?(@.attributes && @.attributes['data-role'] == 'tableRoot')]"
            },
            type: "single",
            templateKey: "customSearchBox"
        }
    ],
    serial: [
        {
            destinationPath: {
                jsonPath: "$..*[?(@.tag == 'thead')]"
            },
            type: "single",
            templateKey: "serialHeader"
        }
    ]
};

const result = runOptions({
    base: myCustomTree,
    options: { showSearch: true, showSerial: true },
    optionDefs: customOptionDefs
});
